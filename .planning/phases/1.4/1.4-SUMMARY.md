## 使用範例

### Webhook 驗籤

```typescript
import { verifyTapPayWebhook } from '@carllee1983/tappay-backend-payment-node'

app.post('/webhook', (req, res) => {
  const isValid = verifyTapPayWebhook(
    req.body,
    req.headers['x-tappay-signature'] as string,
    process.env.TAPPAY_PARTNER_KEY!
  )

  if (!isValid) return res.status(401).json({ error: 'Invalid signature' })

  // 處理 webhook
  res.json({ success: true })
})
```

### 重試機制

```typescript
import { retryWithBackoff, isRetryableError } from '@carllee1983/tappay-backend-payment-node'

const result = await retryWithBackoff(
  async () => client.getTransaction(id),
  {
    maxRetries: 3,
    baseDelay: 100,
    shouldRetry: isRetryableError
  }
)
```

### 安全日誌記錄

```typescript
import { scrubErrorForLogging } from '@carllee1983/tappay-backend-payment-node'

try {
  await client.payByPrime({ prime: 'xxx', amount: 1000 })
} catch (error) {
  // 記錄清理後的錯誤（不包含敏感資訊）
  logger.error('Payment failed:', scrubErrorForLogging(error))
```
